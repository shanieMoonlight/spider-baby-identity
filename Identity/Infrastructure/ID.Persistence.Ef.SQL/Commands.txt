# set one-off env vars for SQL Server migration
$env:ASPNETCORE_ENVIRONMENT = 'Development'
$env:ConnectionStrings__SqlDb = 'Data Source=DESKTOP-A0SH758;Initial Catalog=MyIdDemo;Integrated Security=True;TrustServerCertificate=True;Connection Timeout=30;'

# add migration into the SQL provider project
Add-Migration MyId_Initial   -Context IdSqlMigrationsContext   -Project "Id.Persistence.Ef.SQL"   -StartupProject "MyIdDemo"

# apply the migration to the SQL Server DB
Update-Database   -Context IdSqlMigrationsContext   -Project "Id.Persistence.Ef.SQL"   -StartupProject "MyIdDemo"

# cleanup (optional)
Remove-Item Env:\ConnectionStrings__SqlDb
Remove-Item Env:\ASPNETCORE_ENVIRONMENT
